# Walkthrough of the batch in patches/performance (record-walkthrough.sh), 1440x900. MIDI comes
# from the virtual controller (KABL_PLAYER). Sleeps leave time to listen.
sleep 4
click perform
shot w01-perform-panel
sleep 5
# The sequencer: velocity row and gate mode in its advanced area.
click module:3
click zoom:focus
click toggle:3
shot w02-seq-advanced
sleep 3
# Velocity: step 2 (a quiet one) to full, then back.
dragby knob:3.v2 0 -80
shot w03-velocity-up
sleep 5
key ctrl+z
# Gate: CLOCK (the original, half the step) against LENGTH (35 %), then back.
click sel:3.gate_mode.0
shot w04-clock-gates
sleep 5
key ctrl+z
shot w05-length-gates
sleep 4
click zoom:fit
# Pin the lead's filter cutoff from the module menu, move it left.
rclick module:22
click menu:pin
click menu:pin:cutoff_hz
shot w06-pinned
# Rename it, move it left.
click pmenu:22.cutoff_hz
click prename:22.cutoff_hz
key ctrl+a
type Lead cutoff
key Return
click pmenu:22.cutoff_hz
click pleft:22.cutoff_hz
shot w07-renamed-moved
sleep 2
# MIDI learn: CC 74 from the controller, far below the value: pickup pending.
click plearn:22.cutoff_hz
midi cc 1 74 5
sleep 1
midi on 64 100
shot w08-learned-pickup
sleep 2
# Turning up: nothing moves until the hardware reaches the value, then it takes over.
midi ramp 1 74 5 110 4
sleep 4.5
shot w09-hardware-in-control
midi ramp 1 74 110 60 2
sleep 2.5
midi off 64
sleep 1
# Two knobs at once (the demo's Bass cutoff and Echo feedback on CC 24 and 27), then one undo.
midi cc 1 24 53
midi cc 1 27 60
midi ramp 1 24 53 95 3
midi ramp 1 27 60 90 3
sleep 3.5
shot w10-two-knobs
key ctrl+z
sleep 3
shot w11-one-undo
# Reverb: mix up to fully wet, back with undo.
pan 1060 120 knob:32.mix 700 420
dragby knob:32.mix 0 -150
shot w12-reverb-wet
sleep 5
key ctrl+z
shot w13-reverb-back
sleep 4
# Record, stop the sequencer (the take keeps the tails), stop recording.
click rec-start
sleep 6
shot w14-recording
click prun:1
sleep 7
shot w15-stopped-tails
click rec-stop
sleep 1
shot w16-saved
click prun:1
sleep 4
