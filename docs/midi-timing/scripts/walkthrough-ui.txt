# virtual midi_player fed by play_events.py (scripted MIDI, not a physical controller). The
# app records its own output (Perform > Record); the video's sound is that take.
sleep 2
click rec-start
shot w01-split-open

click toggle:1

sleep 1

dragby knob:6.channel 0 120
sleep 1
shot w04-bass-all-channels
wait 49
key ctrl+z
sleep 1
shot w05-bass-back-to-ch2
wait 52
click rec-stop
sleep 1
