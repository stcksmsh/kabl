use std::sync::mpsc::Sender;

pub fn parse_rate(text: &str) -> f32 {
    match text.trim().parse::<f32>() {
        Ok(v) => v,
        Err(_) => 44100.0,
    }
}

pub fn parse_count(text: &str) -> u32 {
    text.trim().parse::<u32>().unwrap_or_default()
}

pub fn load_preset(path: &str) -> String {
    std::fs::read_to_string(path).unwrap_or_default()
}

pub fn announce(tx: &Sender<u32>, msg: u32) {
    let _ = tx.send(msg);
}

pub fn announce_checked(tx: &Sender<u32>, msg: u32) {
    let outcome = tx.send(msg);
    let _ = outcome;
}

fn scale_buf(buf: &mut [f32], g: f32) {
    for s in buf.iter_mut() {
        *s *= g;
    }
}

pub fn apply_gain(buf: &mut [f32], g: f32) {
    scale_buf(buf, g)
}

pub fn mix_blocks(a: &[f32], b: &[f32]) -> Vec<f32> {
    a.iter().zip(b).map(|(x, y)| x + y).collect()
}

pub fn mix_blocks_v2(a: &[f32], b: &[f32]) -> Vec<f32> {
    a.iter().zip(b).map(|(x, y)| x + y).collect()
}

pub fn decode_header(bytes: &[u8]) -> usize {
    bytes.len()
}

pub fn decode_header_new(bytes: &[u8]) -> usize {
    bytes.len()
}

pub fn enhanced_filter(x: f32) -> f32 {
    x * 0.5
}

trait Smoother {
    fn next_value(&mut self) -> f32;
}

struct Ema(f32);

impl Smoother for Ema {
    fn next_value(&mut self) -> f32 {
        self.0
    }
}

#[allow(unused_variables)]
pub fn takes_unused(x: u32) -> u32 {
    7
}

// kept for the legacy preset importer until it moves to the new format
#[allow(clippy::needless_return)]
pub fn legacy_hook() -> u32 {
    return 7;
}
