import json

def read_config(path):
    """Reads the config."""
    try:
        with open(path) as f:
            return json.load(f)
    except Exception:
        return None


def total(values):
    # Initialize the total to zero
    acc = 0
    for v in values:
        acc += v  # add v to the total
    return acc
